def train_classifier():
    blocks = ["optimiser", "weight_decay", "schedule", "warmup_epochs", "backbone_lr_scale", "sampler",
              "loss_weights", "select"]
    return manifest(
        "train_classifier", "Train the classifier", "train",
        "Trains the classifier on the train part, keeps every epoch, and picks the best epoch on validation.",
        "tools/train.py through the flow: the optimiser (adam or adamw, weight decay only with adamw), the "
        "schedule (constant or cosine, after warm-up epochs), a learning-rate scale for the backbone, the "
        "sampler, the loss weights and the measure that picks the kept epoch on validation. Every epoch's "
        "weights are kept, epoch 0 included, so the witness frames are drawn at every checkpoint. On the "
        "experiment route training and evaluation run in one GPU job (grid/experiment-driver.sbatch: 8 "
        "CPUs, 64 GB) whose wall time is hours.",
        "train",
        inputs=[port_in("model", {"kind": ["model"], "rank": [2], "needs_parts": ["train", "validation"]},
                        "A 2-D classifier built for split data.")],
        outputs=[port_out("checkpoint", "model", [["kind", "checkpoint"], ["set", "trained", True]],
                          "The trained weights of every epoch and the kept one.")],
        params=[
            choice("optimiser", ["adam", "adamw"], "adam", True, "the optimiser"),
            number("weight_decay", 0, 1, 0.0, True,
                   "the weight decay, above 0 only with adamw; src/schwingnet/blocks.py sets no upper bound, "
                   "1 is the sandbox's cap"),
            choice("schedule", ["constant", "cosine"], "constant", True, "the learning rate's schedule"),
            integer("warmup_epochs", 0, 100000, 0, True, "whole epochs of linear warm-up"),
            number("backbone_lr_scale", 0, 1, 1.0, True, "the backbone's learning rate over the head's, above 0 "
                                                         "and at most 1", above=True),
            choice("sampler", ["uniform", "balanced"], "uniform", True, "how training batches are drawn"),
            choice("loss_weights", ["none", "inverse-frequency"], "none", True, "the level loss's class weights"),
            choice("select", ["accuracy", "balanced-accuracy"], "accuracy", True,
                   "the validation measure that picks the kept epoch"),
            integer("epochs", 1, 200, 20, True, "epochs to train (the experiment route takes 1 to 200)"),
            integer("seed", 0, 999999999, 1, True, "the run's seed"),
            number("hours", 0.1, 48, 4.0, False, "the GPU job's wall time, at most 48 hours"),
        ],
        rules=[rule({"weight_decay": {"above": 0}}, {"optimiser": {"in": ["adamw"]}},
                    "weight_decay above 0 needs optimiser=adamw")],
        reads=["train", "validation"],
        job_=job("gpu", "@param.hours", 8, 64),
        deterministic="seeded", seed_param="seed",
        nondeterminism="torch's deterministic kernels are asked for with warn_only, so a GPU run may differ "
                       "in its last bits (tools/train.py).",
        generic_=generic([2], "no", why="The trainer reads 2-D crops of the catalog (tools/train.py); volumes "
                                        "and clips would need a reader and a network of their own."),
        resolve="a trained {family} whose {rank}-D stem takes {sizes.C} channel(s) of {sizes.H} x {sizes.W}; head of {classes}",
        needs=["python", "numpy", "torch", "torchvision", "metaflow"],
        implemented_by=["tools/train.py", "tools/train_blocks.py", "tools/flow.py", "flows/experiment.py",
                        "grid/experiment-driver.sbatch"],
        today_=today("live", route="experiment", word="experiment <issue>", flow="ExperimentFlow",
                     note="one GPU job a request, on the owner's word"),
        experiment={"epochs": "@param.epochs", "seed": "@param.seed", "hours": "@param.hours",
                    "blocks": {name: f"@param.{name}" for name in blocks}})


def train_annealer():
    return manifest(
        "train_annealer", "Train the annealer", "train",
        "One auto-anatomist trial: trains while the mask input is annealed away, one array task a trial.",
        "One trial of the auto-anatomist screen (#483): the anneal schedule (lambda-then-w, joint, "
        "w-first, or fixed, the null), the objective (the oracle's landmarks, or the mask channels "
        "themselves), the naming term, the calibration of the spreads, the epochs, whole phantoms a batch "
        "and the seed. Each trial is one task of a GPU array (grid/autoanatomist.sbatch there: 8 CPUs, 64 "

        params=[
            choice("schedule", ["lambda-then-w", "joint", "w-first", "fixed"], "lambda-then-w", True,
                   "how lambda's and w's ramps are placed; fixed is the null"),
            choice("objective", ["landmarks", "mask"], "landmarks", True,
                   "what the decoder is trained to give"),
            choice("naming", ["on", "off"], "on", True, "whether the naming term is in the loss"),
            choice("calibrate", ["none", "scale", "quantile"], "none", True, "the calibration of the spreads"),
