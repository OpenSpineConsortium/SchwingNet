def split_person():
    return manifest(
        "split_person", "Split by person", "split",
        "Puts each person in train, validation or test by a salted hash; the frozen test persons stay test.",
        "The catalog's split rule (src/schwingnet/catalog.py, data/splits/rule.json): a person on "
        "data/splits/test-persons.json is test; any other person is validation where the first 8 bytes of "
        "sha256 over the salt and the person fall below validation_share of 2**64, training otherwise. "
        "Scans travel with their person, so no person is on two sides, and the split check holds every "
        "catalog version to the rule. Today the catalog step does it when it builds a version.",
        "split",
        inputs=[port_in("data", {"kind": ["data"]}, "Data with a person key.")],
        outputs=[port_out("data", "data", [["parts", list(PARTS)]],
                          "The same data with each person in train, validation or test.")],
        params=[
            number("validation_share", 0.001, 0.999, 0.1, False,
                   "the share of persons hashed into validation (data/splits/rule.json: 0.1)"),
            choice("salt", ["schwingnet-2026-10-21"], "schwingnet-2026-10-21", False,
                   "the salt of the frozen rule; a new salt is a new split"),
        ],
        generic_=generic(ALL_RANKS),
        needs=["python"],
        implemented_by=["src/schwingnet/catalog.py", "tools/catalog.py", "flows/catalog.py", "data/splits/rule.json",
                        "data/splits/test-persons.json"],
        today_=today("flow", flow="CatalogFlow", note="the catalog step splits every version it builds"),
        experiment=None)

def augment():
    return manifest(
        "augment", "Augment the training crops", "augment",
        "Random changes to the training crops each epoch: box jitter, intensity, blur, noise, or the strong set.",
        "The augmenters of tools/train_blocks.py, applied in their fixed order (AUGMENT_ORDER) whatever "
        "order they are named in: box-jitter, intensity, blur, noise, and strong, SyntheX's strong "
        "randomisation (#252). They change the training crops only; validation and test are read as "
        "they are. None named is version 1's training.",

def classifier():
    return manifest(
        "classifier", "Level classifier network", "model",
        "A 2-D image classifier with a 27-way level head; grey crops are tiled to 3 channels for its stem.",
        "One of the eight backbones of tools/train_blocks.py ARCHITECTURES: torchvision's resnet18, "
        "resnet34, resnet50, mobilenet-v3-small and efficientnet-b0, MONAI's densenet121, se-resnet50 and "
        "se-resnext50. The head has 27 outputs, the LEVELS of src/schwingnet/catalog.py, whatever the "
        "data's classes. Grey crops are repeated to 3 channels for the stem, and adaptive pooling lets "
        "any crop size through. ImageNet weights come from a file, never a download; the grid holds only "
        "resnet18's file, so the experiment route runs resnet18 alone (grid/experiment-driver.sbatch).",
        "network",
        inputs=[port_in("data", {"kind": ["data"], "rank": [2], "time": "no", "has_axes": ["C"],
                                 "targets": ["level"]}, "2-D crops with level labels.")],
        outputs=[port_out("model", "data", [["kind", "model"], ["set", "family", "@param.architecture"],
                                            ["set", "classes", 27]],
                          "The network, shaped by the crops it will read.")],
        params=[
            choice("architecture", ARCHITECTURES, "resnet18", True, "the backbone (src/schwingnet/blocks.py)"),
            choice("weights", ["imagenet", "none"], "imagenet", True,
                   "imagenet: the backbone's ImageNet weights from a file (--weights); none: untrained"),
        ],
        deterministic="declared", nondeterminism=RANDOM_HEAD,
        generic_=generic([2], "no", why="2-D torchvision and MONAI networks today (spatial_dims 2, in_channels "
                                        "3), with grey tiled to 3 channels and a head fixed at the 27 levels; a "
                                        "volume needs a 3-D network (classifier_3d), and other classes need the "
                                        "head read from the data."),
        resolve="a {family} whose {rank}-D stem takes {sizes.C} channel(s) of {sizes.H} x {sizes.W}; head of {classes}",
        needs=["python", "torch", "torchvision", "monai"],
        implemented_by=["tools/train_blocks.py", "tools/train.py", "src/schwingnet/blocks.py",
                        "grid/experiment-driver.sbatch"],
        today_=today("live", route="experiment", note="resnet18 with ImageNet weights alone on the grid"),
        experiment={"blocks": {"architecture": "@param.architecture"},
                    "only": {"architecture": ["resnet18"], "weights": ["imagenet"]}})
