"""Writes every DATA file: the contract, the block manifests and the templates."""

import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import gen_contract  # noqa: E402
import gen_templates  # noqa: E402
from gen_blocks_model import MODEL  # noqa: E402
from gen_blocks_prepare import PREPARE  # noqa: E402
from gen_blocks_sources import SOURCES  # noqa: E402
from gen_blocks_train import TRAIN  # noqa: E402
from gen_common import SANDBOX, write_json  # noqa: E402

BLOCKS = SOURCES + PREPARE + MODEL + TRAIN


def main():
    write_json(os.path.join(SANDBOX, "contract.json"), gen_contract.contract())
    ids = []
    for make in BLOCKS:
        value = make()
        ids.append(value["id"])
        write_json(os.path.join(SANDBOX, "blocks", value["id"] + ".json"), value)
    assert len(ids) == len(set(ids)), "duplicate block ids"
    templates = gen_templates.write_all()
    print(len(ids), "blocks;", len(templates), "templates")


if __name__ == "__main__":
    main()
