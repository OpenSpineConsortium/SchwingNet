function clone(v) {
  return JSON.parse(JSON.stringify(v));
}

function withBlocks(extra, contractChange) {
  var d = clone(D);
  extra.forEach(function (m) { d.blocks.push(m); });
  if (contractChange) contractChange(d.contract);
  return d;
}
