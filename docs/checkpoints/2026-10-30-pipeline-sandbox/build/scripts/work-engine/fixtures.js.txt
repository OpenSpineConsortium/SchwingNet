function job(cls, hours, extra) {
  var j = {"class": cls || "none", hours: hours === undefined ? 0 : hours, with_upstream: false, array: false,
    cpus: cls && cls !== "none" ? 4 : 0, memory_gb: cls && cls !== "none" ? 16 : 0};
  Object.keys(extra || {}).forEach(function (k) { j[k] = extra[k]; });
  return j;
}

function today(status, extra) {
  var t = {status: status, route: "", word: "", flow: "", note: ""};
  Object.keys(extra || {}).forEach(function (k) { t[k] = extra[k]; });
  return t;
}

var EVERYTHING = {rank: [0, 1, 2, 3], time: "any", channels: "any", sizes: "any", why: ""};

function manifest(id, stage, glyph, fields) {
  var m = {
    block: 1, id: id, version: 1, title: fields.title || id.replace(/_/g, " "), stage: stage,
    summary: "The " + id + " block of the scratch tests.", does: "It does what the " + id + " test needs.", glyph: glyph,
    inputs: [], outputs: [], params: [], rules: [], reads: {parts: []}, job: job(), deterministic: "exact",
    seed_param: "", nondeterminism: "", generic: null, resolve: null, needs: ["python"], code: "existing",
    implemented_by: ["tools/train.py"], today: today("live", {route: "experiment"}), experiment: {}, card: null
  };
  Object.keys(fields).forEach(function (k) { m[k] = fields[k]; });
  return m;
}

function experiment() {
  return JSON.parse(fs.readFileSync(path.join(__dirname, "experiment.json"), "utf8"));
}

function data(extra) {
  var d = {contract: contract, blocks: blocks, templates: [], experiment: experiment()};
  Object.keys(extra || {}).forEach(function (k) { d[k] = extra[k]; });
  return d;
}

function node(id, block, params, vary) {
  var n = {id: id, block: block, version: 1, params: params || {}};
  if (vary) n.vary = vary;
  return n;
}

function edge(from, outPort, to, inPort) {
  return {from: from, out: outPort, to: to, "in": inPort};
}

// The experiment chain of DATA's experiment_v1, from the scratch blocks.
function chain(name) {
  return {
    sandbox: 1, name: name || "chain",
    nodes: [node("n1", "catalog_v1"), node("n2", "crop_boxes"), node("n3", "normalise"), node("n4", "augment"),
      node("n5", "classifier"), node("n6", "train_classifier"), node("n7", "evaluate_heldout"), node("n8", "report_claim")],
    edges: [edge("n1", "data", "n2", "data"), edge("n2", "data", "n3", "data"), edge("n3", "data", "n4", "data"),
      edge("n4", "data", "n5", "data"), edge("n5", "model", "n6", "model"), edge("n6", "checkpoint", "n7", "checkpoint"),
      edge("n7", "metrics", "n8", "results")],
    layout: {n1: [40, 40]}
  };
}

module.exports = {contract: contract, blocks: blocks, data: data, node: node, edge: edge, chain: chain, manifest: manifest,
