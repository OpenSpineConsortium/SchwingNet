"use strict";

var T = require("./t.js");
var F = require("./fixtures.js");
var SB = T.SB;

function clone(v) {
  return JSON.parse(JSON.stringify(v));
}

function pairs(problems) {
  return Array.from(new Set(problems.map(function (p) { return p.code + "|" + (p.field === null ? "-" : p.field); }))).sort();
}

function check(m, contract) {
  return SB.checkManifest(m, contract || F.contract);
}

function byId(id) {
  return clone(F.blocks.filter(function (m) { return m.id === id; })[0]);
}

var good = byId("train_classifier");
var source = byId("catalog_v1");

  "trial_array", "gpu_render", "gpu_follow", "picky"].forEach(function (id) {
  T.eq(check(byId(id)), [], id + " passes");
});
T.ok(check(good).every(function (p) { return Object.keys(p).join() === "code,field,text"; }), "problems are {code, field, text}");

function expect(change, wanted, what, base) {
  var m = clone(base || good);
  change(m);
  var got = pairs(check(m));
  T.eq(got, wanted.slice().sort(), what);
  return got;
}
