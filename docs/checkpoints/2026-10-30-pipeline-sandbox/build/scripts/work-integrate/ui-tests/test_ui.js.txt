// The sandbox page under real pointer events (Playwright, Chromium). Each check prints PASS or
// FAIL with what it saw; the run ends with the console errors and the timings.

const results = [];
function check(name, ok, seen) {
  results.push({name: name, ok: !!ok, seen: seen});
  console.log((ok ? "PASS " : "FAIL ") + name + (seen !== undefined ? "  " + JSON.stringify(seen) : ""));
}
const PAGE = process.env.PAGE || "page.html";
const sleep = function (ms) { return new Promise(function (r) { setTimeout(r, ms); }); };

async function cardBox(page, id) {
  return page.evaluate(function (id) {
    const e = document.querySelector('.ncard[data-id="' + id + '"]');
    if (!e) return null;
    const r = e.getBoundingClientRect();
    return {x: r.left, y: r.top, w: r.width, h: r.height, right: r.right, bottom: r.bottom};
  }, id);
}
async function graph(page) { return page.evaluate(function () { return window.sandbox.graph(); }); }
async function ghostState(page) {
  return page.evaluate(function () {
    const g = document.getElementById("sb-ghost"), s = document.getElementById("sb-strip");
    const cur = Array.from(s.querySelectorAll(".sug")).findIndex(function (b) { return b.getAttribute("aria-current") === "true"; });
    return {
      ghost: !g.hidden, none: g.classList.contains("none"), strip: !s.hidden, title: document.getElementById("sb-ghost-title").textContent,
      current: cur, entries: Array.from(s.querySelectorAll(".sug")).filter(function (b) { return !b.hidden; }).map(function (b) { return b.querySelector(".stitle").textContent; })
    };
  });
}
async function clickChip(page, title) {
  await page.evaluate(function () { document.getElementById("sb-search").value = ""; });
  await page.fill("#sb-search", title);
  await page.click('.bchip:has-text("' + title + '")');
  await page.fill("#sb-search", "");
  // A person's click on a chip leaves the focus on the chip; clearing the filter here must not leave it in the box.
  await page.evaluate(function () { if (document.activeElement) document.activeElement.blur(); });
}
